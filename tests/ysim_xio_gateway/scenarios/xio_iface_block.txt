#       controller）；send_aux_data_updates 关闭原生 AuxData 推送
#       （客户端未注册 ID 73 包，避免未知包告警噪声）。
# 注意：pause_for_connection 下无客户端时 mission 将等待，勿用于无人值守场景。
xio_interface
   multicast 235.7.11.27 192.168.
   port 3002
   heartbeat_interval 1.0 s
   time_to_live 8
   pause_for_connection true
   send_aux_data_updates false
